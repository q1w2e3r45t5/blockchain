// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;
contract BalanceTransferContract {
    address public owner;
    event Deposited(address indexed sender, uint256 amount);
    event Transferred(address indexed recipient, uint256 amount);
    modifier onlyOwner() {
        require(msg.sender == owner, "Not the owner");
        _;
    }
    constructor() {
        owner = msg.sender;
    }
    receive() external payable {
        emit Deposited(msg.sender, msg.value);
    }
    function deposit() external payable {
        require(msg.value > 0, "Amount must be greater than 0");
        emit Deposited(msg.sender, msg.value);
    }
    function getContractBalance() external view returns (uint256) {
        return address(this).balance;
    }
    function transferBalance(address payable _to, uint256 _amount)
        external onlyOwner {
        require(_to != address(0), "Invalid address");
        require(address(this).balance >= _amount, "Insufficient balance");
        (bool success, ) = _to.call{value: _amount}("");
        require(success, "Transfer failed");
        emit Transferred(_to, _amount);
    }
    function transferAll(address payable _to) external onlyOwner {
        require(_to != address(0), "Invalid address");
        uint256 total = address(this).balance;
        require(total > 0, "Balance is zero");
        (bool success, ) = _to.call{value: total}("");
        require(success, "Transfer failed");
        emit Transferred(_to, total);
    }
}

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;
contract SimpleToken {
    string public name = "EduToken";
    string public symbol = "EDU";
    uint8 public decimals = 18;
    uint256 public totalSupply;
    mapping(address => uint256) public balanceOf;
    event Transfer(address indexed from, address indexed to, uint256 value);
    constructor(uint256 initialSupply) {
        totalSupply = initialSupply * 10 ** uint256(decimals);
        balanceOf[msg.sender] = totalSupply;
        emit Transfer(address(0), msg.sender, totalSupply);
    }
    function transfer(address to, uint256 value) public returns (bool) {
        require(to != address(0), "Invalid recipient");
        require(balanceOf[msg.sender] >= value, "Not enough tokens");
        balanceOf[msg.sender] -= value;
        balanceOf[to] += value;
        emit Transfer(msg.sender, to, value);
        return true;
    }
}

import hashlib
import time

def sha(data):
    return hashlib.sha256(data.encode()).hexdigest()

def merkle(tx):
    h = [sha(t) for t in tx]
    while len(h) > 1:
        if len(h) % 2:
            h.append(h[-1])
        h = [sha(h[i] + h[i+1]) for i in range(0, len(h), 2)]
    return h[0]

class Block:
    def __init__(self, index, prev, tx):
        self.index = index
        self.prev = prev
        self.tx = tx
        self.root = merkle(tx)
        self.time = time.time()
        self.hash = self.calculate()

    def calculate(self):
        return sha(str(self.index) + self.prev + self.root + str(self.time))

class Blockchain:
    def __init__(self):
        self.chain = [Block(0, "0", ["Genesis"])]

    def add(self, tx):
        self.chain.append(Block(len(self.chain), self.chain[-1].hash, tx))

    def valid(self):
        for i, b in enumerate(self.chain):
            prev = "0" if i == 0 else self.chain[i-1].hash
            if b.prev != prev or b.root != merkle(b.tx) or b.hash != b.calculate():
                return False
        return True

    def display(self):
        for b in self.chain:
            print("\nBlock:", b.index)
            print("Previous Hash:", b.prev)
            print("Merkle Root:", b.root)
            print("Block Hash:", b.hash)
            print("Transactions:", b.tx)

bc = Blockchain()
bc.add(["Alice -> Bob: 5", "Bob -> Carol: 2"])
bc.add(["Eve -> Frank: 3", "George -> Harry: 4"])

bc.display()
